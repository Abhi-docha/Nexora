rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // =========================================================================
    // NEXORA - Enterprise Operations Cloud
    // Authoritative Zero-Trust Multi-Tenant Security Rules (V3 - Hardened AI)
    //
    // KEY PROTECTIONS:
    // 1. Strict Multi-User & Tenant Partitioning: Zero data cross-talk or leakage.
    // 2. Dedicated AI Subsystems: AI chat history, consent tokens, staged proposals,
    //    and immutable tamper-proof audit trails.
    // 3. User-Governed AI Mutations: AI proposals require explicit authenticated sign-off.
    // 4. Safe evaluation syntax: No unsupported regex flags like (?i) that crash rules.
    // 5. High-concurrency design: Path-isolated writes eliminate locking & lag.
    // =========================================================================

    // Default deny all access
    match /{document=**} {
      allow read, write: if false;
    }

    // Health / connectivity check
    match /test/{testId} {
      allow read: if true;
    }

    // =========================================================================
    // PLATFORM ADMIN CONFIGURATION
    // Verified against authenticated UID and primary platform owner email.
    // Safe case checking without unsupported regex flags.
    // =========================================================================
    function isPlatformAdmin() {
      return isSignedIn() && (
        request.auth.uid == "0FQhenLKBbdNW6cXYNgenoWYahO2" ||
        request.auth.uid == "REPLACE_WITH_REAL_PLATFORM_ADMIN_UID" ||
        request.auth.uid == "YOUR_PLATFORM_ADMIN_UID_1" ||
        (request.auth.token.email != null && 
         (request.auth.token.email == "abhi30382.AK@gmail.com" || 
          request.auth.token.email == "abhi30382.ak@gmail.com" ||
          request.auth.token.email.lower() == "abhi30382.ak@gmail.com")) ||
        exists(/databases/$(database)/documents/platform/admins/users/$(request.auth.uid))
      );
    }

    // =========================================================================
    // CORE SECURITY PRIMITIVES & HELPERS
    // =========================================================================
    function isSignedIn() {
      return request.auth != null && request.auth.uid != null;
    }

    function isOwner(userId) {
      return isSignedIn() && request.auth.uid == userId;
    }

    function incoming() {
      return request.resource.data;
    }

    function existing() {
      return resource.data;
    }

    function isValidId(id) {
      return id is string && id.size() > 0 && id.size() <= 128 && id.matches('^[a-zA-Z0-9_\\-]+$');
    }

    // Multi-tenant Org Member Verification
    function isOrgMember(orgId) {
      return isSignedIn() && (
        isPlatformAdmin() ||
        exists(/databases/$(database)/documents/organizations/$(orgId)/members/$(request.auth.uid)) ||
        (exists(/databases/$(database)/documents/organizations/$(orgId)) &&
         get(/databases/$(database)/documents/organizations/$(orgId)).data.ownerUid == request.auth.uid)
      );
    }

    function getMemberDoc(orgId) {
      return get(/databases/$(database)/documents/organizations/$(orgId)/members/$(request.auth.uid)).data;
    }

    function isOrgOwner(orgId) {
      return isSignedIn() && (
        isPlatformAdmin() ||
        (exists(/databases/$(database)/documents/organizations/$(orgId)) &&
         get(/databases/$(database)/documents/organizations/$(orgId)).data.ownerUid == request.auth.uid) ||
        (exists(/databases/$(database)/documents/organizations/$(orgId)/members/$(request.auth.uid)) &&
         getMemberDoc(orgId).role == 'OWNER')
      );
    }

    function isOrgAdmin(orgId) {
      return isSignedIn() && (
        isPlatformAdmin() ||
        isOrgOwner(orgId) ||
        (exists(/databases/$(database)/documents/organizations/$(orgId)/members/$(request.auth.uid)) &&
         (getMemberDoc(orgId).role == 'OWNER' || getMemberDoc(orgId).role == 'ORG_ADMIN'))
      );
    }

    function isOrgManager(orgId) {
      return isSignedIn() && (
        isOrgAdmin(orgId) ||
        (exists(/databases/$(database)/documents/organizations/$(orgId)/members/$(request.auth.uid)) &&
         (getMemberDoc(orgId).role == 'MANAGER' || getMemberDoc(orgId).role == 'PROJECT_MANAGER' || getMemberDoc(orgId).role == 'TEAM_LEAD'))
      );
    }

    function isUserActive(orgId) {
      return isPlatformAdmin() ||
        (exists(/databases/$(database)/documents/organizations/$(orgId)) &&
         get(/databases/$(database)/documents/organizations/$(orgId)).data.ownerUid == request.auth.uid) ||
        (exists(/databases/$(database)/documents/organizations/$(orgId)/members/$(request.auth.uid)) &&
         getMemberDoc(orgId).status == 'ACTIVE');
    }

    // =========================================================================
    // USERS COLLECTION: /users/{userId}
    // Application user profile. Normal users can read and write only their own profile.
    // Platform administrators can inspect profiles.
    // =========================================================================
    match /users/{userId} {
      allow get: if isOwner(userId) || isPlatformAdmin();
      allow list: if isPlatformAdmin() || (isSignedIn() && resource.data.uid == request.auth.uid);

      allow create: if isOwner(userId) &&
        isValidId(userId) &&
        incoming().uid == request.auth.uid &&
        incoming().email is string && incoming().email.size() <= 200 &&
        incoming().accountStatus in ['ACTIVE', 'SUSPENDED'] &&
        (!('isPlatformAdmin' in incoming()) || incoming().isPlatformAdmin == false);

      allow update: if (isOwner(userId) &&
        incoming().uid == existing().uid &&
        incoming().email == existing().email &&
        (!('isPlatformAdmin' in incoming()) || incoming().isPlatformAdmin == existing().isPlatformAdmin) &&
        (!('accountStatus' in incoming()) || incoming().accountStatus == existing().accountStatus)
      ) || isPlatformAdmin();

      allow delete: if isPlatformAdmin();

      match /preferences/{prefId} {
        allow read, write: if isOwner(userId) || isPlatformAdmin();
      }

      // -----------------------------------------------------------------------
      // AI DATA ACCESS CONSENT (STRICT USER-LEVEL PRIVACY GATE)
      // Stores user explicit authorization grants and revocations.
      // Strictly private: User A can NEVER view or alter User B's consent.
      // -----------------------------------------------------------------------
      match /ai_consent/{consentId} {
        allow read, write: if isOwner(userId) || isPlatformAdmin();
      }

      // -----------------------------------------------------------------------
      // AI CHAT SESSIONS & CONVERSATION LOGS (MULTI-USER CONCURRENCY ISOLATION)
      // Stores isolated chat history. Each user writes to their own document tree.
      // Eliminates locking lag, guarantees zero cross-user message contamination.
      // -----------------------------------------------------------------------
      match /ai_sessions/{sessionId} {
        allow read, write: if isOwner(userId) || isPlatformAdmin();

        match /messages/{messageId} {
          allow read, write: if isOwner(userId) || isPlatformAdmin();
        }
      }

      match /ai_chats/{chatId} {
        allow read, write: if isOwner(userId) || isPlatformAdmin();

        match /messages/{messageId} {
          allow read, write: if isOwner(userId) || isPlatformAdmin();
        }
      }
    }

    // =========================================================================
    // ORGANIZATIONS COLLECTION: /organizations/{orgId}
    // Tenant boundaries are enforced strictly by path and membership.
    // =========================================================================
    match /organizations/{orgId} {
      allow get: if isOrgMember(orgId) || isPlatformAdmin();
      allow list: if isPlatformAdmin();

      allow create: if isSignedIn() &&
        isValidId(orgId) &&
        incoming().id == orgId &&
        incoming().ownerUid == request.auth.uid &&
        incoming().name is string && incoming().name.size() > 0 && incoming().name.size() <= 150 &&
        incoming().slug is string && incoming().slug.size() > 0 && incoming().slug.size() <= 60 &&
        incoming().planId == 'FREE' &&
        incoming().status in ['ACTIVE', 'SUSPENDED', 'ARCHIVED'] &&
        (!('dodoCustomerId' in incoming())) &&
        (!('dodoSubscriptionId' in incoming()));

      allow update: if (isOrgOwner(orgId) &&
        incoming().id == existing().id &&
        incoming().ownerUid == existing().ownerUid &&
        incoming().planId == existing().planId &&
        (!('billingStatus' in incoming()) || incoming().billingStatus == existing().billingStatus) &&
        (!('subscriptionStatus' in incoming()) || incoming().subscriptionStatus == existing().subscriptionStatus) &&
        (!('dodoCustomerId' in incoming()) || incoming().dodoCustomerId == existing().dodoCustomerId) &&
        (!('dodoSubscriptionId' in incoming()) || incoming().dodoSubscriptionId == existing().dodoSubscriptionId) &&
        (!('billing' in incoming()) || incoming().billing == existing().billing)
      ) || isPlatformAdmin();

      allow delete: if isPlatformAdmin();

      // -----------------------------------------------------------------------
      // MEMBERS: /organizations/{orgId}/members/{memberUid}
      // -----------------------------------------------------------------------
      match /members/{memberUid} {
        allow read: if isOrgMember(orgId);

        allow create: if isOrgAdmin(orgId) || (
          isOwner(memberUid) && (
            (request.resource.data.role == 'OWNER' &&
             get(/databases/$(database)/documents/organizations/$(orgId)).data.ownerUid == request.auth.uid) ||
            exists(/databases/$(database)/documents/organizations/$(orgId)/invites/$(request.resource.data.inviteToken))
          )
        );

        allow update: if isOrgAdmin(orgId) && (
          request.auth.uid != memberUid || isOrgOwner(orgId) || isPlatformAdmin()
        );

        allow delete: if (isOrgAdmin(orgId) && existing().role != 'OWNER') || isPlatformAdmin();
      }

      // -----------------------------------------------------------------------
      // INVITATIONS: /organizations/{orgId}/invites/{inviteId}
      // -----------------------------------------------------------------------
      match /invites/{inviteId} {
        allow read: if isOrgMember(orgId);
        allow create, update, delete: if isOrgAdmin(orgId);
      }

      // -----------------------------------------------------------------------
      // AI PROPOSAL STAGING (ZERO SILENT MUTATIONS GUARD)
      // Stores AI generated Diff Cards requiring explicit human review.
      // -----------------------------------------------------------------------
      match /ai_proposals/{proposalId} {
        allow read: if isOrgMember(orgId) && (
          resource == null ||
          (!('proposedForUid' in resource.data)) ||
          resource.data.proposedForUid == request.auth.uid ||
          isOrgManager(orgId) ||
          isPlatformAdmin()
        );

        allow create: if isOrgMember(orgId) &&
          isValidId(proposalId) &&
          incoming().orgId == orgId &&
          incoming().proposedForUid == request.auth.uid;

        allow update: if isOrgMember(orgId) && (
          isPlatformAdmin() ||
          isOrgManager(orgId) ||
          (resource != null && 'proposedForUid' in resource.data && resource.data.proposedForUid == request.auth.uid)
        );

        allow delete: if isOrgAdmin(orgId) || isPlatformAdmin() || (
          isOrgMember(orgId) && resource != null && 'proposedForUid' in resource.data && resource.data.proposedForUid == request.auth.uid
        );
      }

      // -----------------------------------------------------------------------
      // AI AUDIT LOGS (IMMUTABLE ENTERPRISE RECORD)
      // Records every AI query, data access grant, and approved action.
      // -----------------------------------------------------------------------
      match /ai_audit_logs/{logId} {
        allow read: if isOrgAdmin(orgId) || isPlatformAdmin();
        allow create: if isOrgMember(orgId) &&
          isValidId(logId) &&
          incoming().orgId == orgId &&
          incoming().actorUid == request.auth.uid;
        allow update, delete: if false; // Append-only immutable log
      }

      // -----------------------------------------------------------------------
      // PROJECTS: /organizations/{orgId}/projects/{projectId}
      // -----------------------------------------------------------------------
      match /projects/{projectId} {
        allow read: if isOrgMember(orgId);

        allow create: if isOrgMember(orgId) && (
          isPlatformAdmin() ||
          isOrgAdmin(orgId) ||
          (
            isUserActive(orgId) &&
            isValidId(projectId) &&
            (!('orgId' in incoming()) || incoming().orgId == orgId) &&
            incoming().name is string && incoming().name.size() > 0 && incoming().name.size() <= 200
          )
        );

        allow update: if isOrgMember(orgId) && (
          isPlatformAdmin() ||
          isOrgAdmin(orgId) ||
          isOrgManager(orgId) ||
          isUserActive(orgId)
        );

        allow delete: if isOrgAdmin(orgId) || isPlatformAdmin();

        match /activities/{actId} {
          allow read, write: if isOrgMember(orgId);
        }
      }

      // -----------------------------------------------------------------------
      // TASKS: /organizations/{orgId}/tasks/{taskId}
      // -----------------------------------------------------------------------
      match /tasks/{taskId} {
        allow read: if isOrgMember(orgId);

        allow create: if isOrgMember(orgId) && (
          isPlatformAdmin() ||
          isOrgAdmin(orgId) ||
          isOrgManager(orgId) ||
          (
            isUserActive(orgId) &&
            isValidId(taskId) &&
            (!('orgId' in incoming()) || incoming().orgId == orgId) &&
            incoming().title is string && incoming().title.size() > 0 && incoming().title.size() <= 250
          )
        );

        allow update: if isOrgMember(orgId) && (
          isPlatformAdmin() ||
          isOrgAdmin(orgId) ||
          isOrgManager(orgId) ||
          isUserActive(orgId)
        );

        allow delete: if isOrgManager(orgId) || isPlatformAdmin() || (
          isOrgMember(orgId) && resource != null && 'reporterUid' in resource.data && resource.data.reporterUid == request.auth.uid
        );
      }

      // -----------------------------------------------------------------------
      // WORKFLOWS & RUNS: /organizations/{orgId}/workflows/{workflowId}
      // -----------------------------------------------------------------------
      match /workflows/{workflowId} {
        allow read: if isOrgMember(orgId);
        allow create, update: if (isOrgManager(orgId) || isPlatformAdmin()) &&
          isValidId(workflowId) &&
          (!('orgId' in incoming()) || incoming().orgId == orgId);
        allow delete: if isOrgAdmin(orgId) || isPlatformAdmin();
      }

      match /workflowRuns/{runId} {
        allow read: if isOrgMember(orgId);
        allow create, update: if isOrgMember(orgId);
      }

      // -----------------------------------------------------------------------
      // REQUESTS & APPROVALS: /organizations/{orgId}/requests/{requestId}
      // -----------------------------------------------------------------------
      match /requests/{requestId} {
        allow read: if isOrgMember(orgId);

        allow create: if isOrgMember(orgId) && (
          isPlatformAdmin() ||
          isOrgAdmin(orgId) ||
          isOrgManager(orgId) ||
          (
            isUserActive(orgId) &&
            isValidId(requestId) &&
            (!('orgId' in incoming()) || incoming().orgId == orgId) &&
            incoming().title is string && incoming().title.size() <= 200 &&
            (!('requesterUid' in incoming()) || incoming().requesterUid == request.auth.uid)
          )
        );

        // Requester can update draft/cancel; Approvers, Managers, Org Admins, Platform Admins can update/decide
        allow update: if isOrgMember(orgId) && (
          isPlatformAdmin() ||
          isOrgAdmin(orgId) ||
          isOrgManager(orgId) ||
          (resource != null && 'requesterUid' in resource.data && resource.data.requesterUid == request.auth.uid)
        );

        allow delete: if isOrgAdmin(orgId) || isPlatformAdmin() || (
          isOrgMember(orgId) && resource != null && 'requesterUid' in resource.data && resource.data.requesterUid == request.auth.uid
        );
      }

      // -----------------------------------------------------------------------
      // GOALS & OKRs: /organizations/{orgId}/goals/{goalId}
      // -----------------------------------------------------------------------
      match /goals/{goalId} {
        allow read: if isOrgMember(orgId);
        allow create, update: if isOrgMember(orgId) && (
          isPlatformAdmin() ||
          isOrgAdmin(orgId) ||
          isOrgManager(orgId) ||
          isUserActive(orgId)
        );
        allow delete: if isOrgManager(orgId) || isPlatformAdmin();
      }

      // -----------------------------------------------------------------------
      // INCIDENTS: /organizations/{orgId}/incidents/{incidentId}
      // -----------------------------------------------------------------------
      match /incidents/{incidentId} {
        allow read: if isOrgMember(orgId);

        allow create: if isOrgMember(orgId) && (
          isPlatformAdmin() ||
          isOrgAdmin(orgId) ||
          isOrgManager(orgId) ||
          (
            isUserActive(orgId) &&
            isValidId(incidentId) &&
            (!('orgId' in incoming()) || incoming().orgId == orgId)
          )
        );

        allow update: if isOrgMember(orgId) && (
          isPlatformAdmin() ||
          isOrgAdmin(orgId) ||
          isOrgManager(orgId) ||
          isUserActive(orgId)
        );

        allow delete: if isOrgAdmin(orgId) || isPlatformAdmin();
      }

      // -----------------------------------------------------------------------
      // CRM / CUSTOMERS: /organizations/{orgId}/customers/{customerId}
      // -----------------------------------------------------------------------
      match /customers/{customerId} {
        allow read: if isOrgMember(orgId);
        allow create, update: if isOrgMember(orgId) &&
          incoming().orgId == orgId &&
          incoming().name is string && incoming().name.size() <= 150;
        allow delete: if isOrgManager(orgId);
      }

      // -----------------------------------------------------------------------
      // VENDORS: /organizations/{orgId}/vendors/{vendorId}
      // -----------------------------------------------------------------------
      match /vendors/{vendorId} {
        allow read: if isOrgMember(orgId);
        allow create, update: if isOrgMember(orgId) &&
          incoming().orgId == orgId &&
          incoming().name is string && incoming().name.size() <= 150;
        allow delete: if isOrgAdmin(orgId);
      }

      // -----------------------------------------------------------------------
      // ASSETS: /organizations/{orgId}/assets/{assetId}
      // -----------------------------------------------------------------------
      match /assets/{assetId} {
        allow read: if isOrgMember(orgId);
        allow create, update: if isOrgMember(orgId) &&
          incoming().orgId == orgId &&
          incoming().name is string && incoming().name.size() <= 150;
        allow delete: if isOrgAdmin(orgId);
      }

      // -----------------------------------------------------------------------
      // KNOWLEDGE BASE: /organizations/{orgId}/knowledgeArticles/{articleId}
      // -----------------------------------------------------------------------
      match /knowledgeArticles/{articleId} {
        allow read: if isOrgMember(orgId);
        allow create, update: if isOrgMember(orgId) &&
          (!('orgId' in incoming()) || incoming().orgId == orgId) &&
          incoming().title is string && incoming().title.size() <= 200;
        allow delete: if isOrgAdmin(orgId) || isPlatformAdmin() || (
          isOrgMember(orgId) && resource != null && 'authorUid' in resource.data && resource.data.authorUid == request.auth.uid
        );
      }

      // -----------------------------------------------------------------------
      // MEETINGS & DECISIONS: /organizations/{orgId}/meetings/{meetingId}
      // -----------------------------------------------------------------------
      match /meetings/{meetingId} {
        allow read: if isOrgMember(orgId);
        allow create, update: if isOrgMember(orgId) &&
          (!('orgId' in incoming()) || incoming().orgId == orgId) &&
          incoming().title is string && incoming().title.size() <= 200;
        allow delete: if isOrgManager(orgId) || isPlatformAdmin();
      }

      // -----------------------------------------------------------------------
      // KPIS: /organizations/{orgId}/kpis/{kpiId}
      // -----------------------------------------------------------------------
      match /kpis/{kpiId} {
        allow read: if isOrgMember(orgId);
        allow create, update: if (isOrgManager(orgId) || isPlatformAdmin()) &&
          (!('orgId' in incoming()) || incoming().orgId == orgId);
        allow delete: if isOrgAdmin(orgId) || isPlatformAdmin();
      }

      // -----------------------------------------------------------------------
      // DASHBOARDS & REPORTS: /organizations/{orgId}/dashboards/{dashId}
      // -----------------------------------------------------------------------
      match /dashboards/{dashId} {
        allow read: if isOrgMember(orgId);
        allow create, update: if isOrgMember(orgId) &&
          (!('orgId' in incoming()) || incoming().orgId == orgId);
        allow delete: if isOrgAdmin(orgId) || isPlatformAdmin() || (
          isOrgMember(orgId) && resource != null && 'ownerUid' in resource.data && resource.data.ownerUid == request.auth.uid
        );
      }

      match /reports/{reportId} {
        allow read: if isOrgMember(orgId);
        allow create, update: if isOrgMember(orgId) &&
          incoming().orgId == orgId;
        allow delete: if isOrgAdmin(orgId);
      }

      // -----------------------------------------------------------------------
      // ACTIVITY EVENTS: /organizations/{orgId}/activityEvents/{eventId}
      // Append-only application activity log
      // -----------------------------------------------------------------------
      match /activityEvents/{eventId} {
        allow read: if isOrgMember(orgId);
        allow create: if isOrgMember(orgId) &&
          isValidId(eventId) &&
          incoming().orgId == orgId &&
          incoming().actorUid == request.auth.uid;
        allow update, delete: if false; // Append-only immutable history
      }

      // -----------------------------------------------------------------------
      // NOTIFICATIONS: /organizations/{orgId}/notifications/{notifId}
      // -----------------------------------------------------------------------
      match /notifications/{notifId} {
        allow read: if isOrgMember(orgId) && (resource == null || resource.data.recipientUid == request.auth.uid);
        allow create: if isOrgMember(orgId);
        allow update: if isOrgMember(orgId) && resource != null && resource.data.recipientUid == request.auth.uid;
        allow delete: if isOrgMember(orgId) && resource != null && resource.data.recipientUid == request.auth.uid;
      }

      // -----------------------------------------------------------------------
      // USAGE & SETTINGS: /organizations/{orgId}/usage/{usageId}
      // -----------------------------------------------------------------------
      match /usage/{usageId} {
        allow read: if isOrgMember(orgId);
        allow write: if isOrgAdmin(orgId) || isPlatformAdmin();
      }

      // -----------------------------------------------------------------------
      // BILLING & SUBSCRIPTION DATA: /organizations/{orgId}/billing/{document=**}
      // Read-only for org members. Writes restricted strictly to Admin SDK or Platform Admins.
      // -----------------------------------------------------------------------
      match /billing/{document=**} {
        allow read: if isOrgMember(orgId);
        allow write: if isPlatformAdmin();
      }

      match /settings/{settingId} {
        allow read: if isOrgMember(orgId);
        allow write: if isOrgAdmin(orgId) || isPlatformAdmin();
      }
    }

    // =========================================================================
    // PLATFORM CONTROL & SUPER ADMIN
    // Restricted strictly to platform administrators
    // =========================================================================
    match /platform/{document=**} {
      allow read, write: if isPlatformAdmin();
    }
  }
}
